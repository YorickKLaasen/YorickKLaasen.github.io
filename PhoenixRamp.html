About the project

Phoenix Ramp is a game about the Phoenix disaster: the fire on the steamship SS Phoenix on Lake Michigan, close to Sheboygan (Wisconsin, USA). The ship was carrying many Dutch emigrants, including people from the Achterhoek, and an estimated 190 people or more lost their lives.

The game was made as part of a school project, to inform students about the incident in a playful way.

Project

Made as an AGP project: Agile Game Production. We worked as a team using agile methods (sprints, task division and regular retrospectives).

Goal

Inform students about the Phoenix disaster through a game, instead of just a text or a presentation.

MiningOres.cs

What this script does: handles the mining and collecting of ores and respawning


using System.Collections;
using UnityEngine;

public class MiningOres : MonoBehaviour
{
    [SerializeField] OreTypesData oreTypesData;
    [SerializeField] HandleOresInventory oreInventory;
    [SerializeField] Sprite originalSprite;
    [SerializeField] Sprite mediumOreSprite;
    [SerializeField] Sprite smallOreSprite;
    [SerializeField] BoxCollider2D boxCollider;
    [SerializeField] SpriteRenderer healthBarSprite;
    private SpriteRenderer spriteRenderer;
    public float blockStrength;
    public float currentBlockStrength;
    public float playerMiningSpeed;
    //[SerializeField] PlayerData playerData;
    Rigidbody2D rigidBody;
    float dropAmount = 0;
    float fortune = 0;
    public float fillAmount = 1.25f;
    [SerializeField] GameObject healthBar1;
    [SerializeField] GameObject healthBar2;
    [SerializeField] ParticleSystem breakingOreParticle;

    void Start()
    {
        oreInventory = GetComponentInParent();
        oreTypesData = GetComponent();
        print(oreTypesData.oreType);
        //Debug.Log("Ore type: " + oreTypesData.oreType);
        //playerMiningSpeed = playerData.miningSpeed;
        rigidBody = GetComponent();
        spriteRenderer = GetComponent();
        boxCollider = GetComponent();
        currentBlockStrength = blockStrength;
    }

    private void Update()
    {
        if (currentBlockStrength != blockStrength)
        {
            healthBar1.SetActive(true);
            healthBar2.SetActive(true);
        }
        fillAmount = currentBlockStrength / blockStrength;
        healthBarSprite.transform.localScale = new Vector3(fillAmount, 0.376375f, 1f);
    }
    private void OnCollisionEnter2D(Collision2D collision)
    {
        if (collision.gameObject.CompareTag("Player"))
        {
            currentBlockStrength -= 20;
            breakingOreParticle.Play();
            BlockHealthUpdate();
        }
    }

    void BlockHealthUpdate()
    {
        if (currentBlockStrength <= blockStrength * 0.50f)
        {
            spriteRenderer.sprite = mediumOreSprite;
        }
        if (currentBlockStrength <= blockStrength * 0.25f)
        {
            spriteRenderer.sprite = smallOreSprite;
        }
        if (currentBlockStrength <= 0)
        {
            print("Ik ben dood");
            if (oreTypesData != null && oreInventory != null)
            {
                oreInventory.AddOreToInventory(oreTypesData);
            }
            StartCoroutine(RespawnOres());
        }
    }
    IEnumerator RespawnOres()
    {
        spriteRenderer.sprite = originalSprite;
        healthBar1.SetActive(false);
        healthBar2.SetActive(false);
        spriteRenderer.enabled = false;
        boxCollider.enabled = false;
        //gameObject.SetActive(false);
        blockStrength = 100;
        currentBlockStrength = blockStrength;
        yield return new WaitForSeconds(60f);
        print("ik leef");
        spriteRenderer.enabled = true;
        boxCollider.enabled = true;
    }
}

PlaceOres.cs

What this script does: spawns ores in random positions on the groundTilemap


using UnityEngine;
using UnityEngine.Tilemaps;

public class PlaceOres : MonoBehaviour
{
    [Header("Tilemap")]
    [SerializeField] private Tilemap groundTilemap;

    [Header("Ore Settings")]
    [SerializeField] private GameObject[] orePrefabs;
    [Range(0f, 1f)][SerializeField] private float spawnChance = 0.1f;

    [Header("Parent")]
    [SerializeField] private Transform oreParent;

    [ContextMenu("Generate Ores")]
    public void GenerateOres()
    {
        ClearOres();

        BoundsInt bounds = groundTilemap.cellBounds;

        for (int x = bounds.xMin; x < bounds.xMax; x++)
        {
            for (int y = bounds.yMin; y < bounds.yMax; y++)
            {
                Vector3Int cellPosition = new Vector3Int(x, y, 0);

                // Alleen als er een tile ligt
                if (!groundTilemap.HasTile(cellPosition)) continue;

                // Kans check
                if (Random.value < spawnChance)
                {
                    Vector3 worldPos = groundTilemap.GetCellCenterWorld(cellPosition);

                    // Random ore kiezen
                    GameObject randomOre = orePrefabs[Random.Range(0, orePrefabs.Length)];

                    Instantiate(randomOre, worldPos, Quaternion.identity, oreParent);
                }
            }
        }
    }

    private void ClearOres()
    {
        for (int i = oreParent.childCount - 1; i >= 0; i--)
        {
            Transform child = oreParent.GetChild(i);

            if (Application.isPlaying)
                Destroy(child.gameObject);
            else
                DestroyImmediate(child.gameObject);
        }
    }
}
