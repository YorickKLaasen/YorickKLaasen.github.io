Project / assignment

Kamo is a roguelike tower climber where you play as Kamo, a samurai duck with a huge shotgun. This game was made as a school project.

Assignment: This project was called Freedom and as such we were given almost complete freedom in what to make. of course there were a few rules we had to follow such as Clean Code, Documents and other things for the rest it was all up to us.

Team: Daniël Freriksen, Rinze Kuizenga, Ruben Stender, Simon Boersma, Luuk van Norden and myself.

What I made

System

Upgrade system

after you beat the tower you can choose a upgrade to start with in the next tower to make it easier to progress. each time you get the choice between 3 powerups so its up to luck to get something good.

UI

UI

The UI are pretty simple and show you what you need to know, Settings simply shows the controls for the game, Pause screen pauses your game so you can take a break or check the controls, Game over screen shows that you died how long you survived and a retry or exit button, And as last Tower progress shows how far you are in the tower until you get the bossfight.

Upgrades.cs 
using UnityEngine;

public class Upgrades : MonoBehaviour
{
    [SerializeField] UpgradeTiers upgradeTiers;
    [SerializeField] Weaponshot weaponShot;//ammo en reload en shotgunPellets
    [SerializeField] PlayerData playerData; //Speed
    [SerializeField] PlayerHp playerHp;
    
    float[] reloadTimes = {0.5f,0.4f,0.25f };
    int[] shotgunPelletAmounts = { 2, 1, 2 };
    float[] speedBoostAmounts = { 1.3f, 1.05f, 1.1f };
    int[] moreDamageAmounts = { 1, 1, 1 };
    int[] ammoAmounts = { 1, 1, 1 };
    public int damageIncrease;

    public void OnReloadClick()
    {
        weaponShot.reloadTime = reloadTimes[upgradeTiers.reloadTier - 1];
    }
    public void OnShotgunPelletClick()
    {
        weaponShot.pelletIncrease += shotgunPelletAmounts[upgradeTiers.pelletTier - 1];
    }
    public void OnSpeedBoostClick()
    {
        playerData.runMaxSpeed *= speedBoostAmounts[upgradeTiers.speedBoostTier - 1];
    }
    public void OnHeartClick()
    {
        if (playerHp.playerHp == 2)
        {
            playerHp.heart1.SetActive(true);
            playerHp.playerHp++;
        }
        else if (playerHp.playerHp == 1)
        {
            playerHp.heart2.SetActive(true);
            playerHp.playerHp++;
        }
    }
    public void OnDamageClick()
    {
        damageIncrease += moreDamageAmounts[upgradeTiers.moreDamageTier - 1];
    }
    public void OnAmmoClick()
    {
        weaponShot.maxAmmo += ammoAmounts[upgradeTiers.ammoTier - 1];
    }
}
Progressbar.cs 
using Unity.VisualScripting;
using UnityEngine;
using UnityEngine.UI;
public class CompletionBarFiller : MonoBehaviour
{
    [SerializeField] Image completionBar;
    public float fill = 0f;
    float fillTime = 3f;
    [SerializeField] Gradient gradient;
    [SerializeField] MapGenerator mapGenerator;
    void Update()
    {
        fill = mapGenerator.roomCounter / 5;
        completionBar.fillAmount = Mathf.MoveTowards(completionBar.fillAmount, fill, Time.deltaTime / fillTime);
        completionBar.color = gradient.Evaluate(completionBar.fillAmount);
    }
}
Flashlight.cs 
using UnityEngine;

public class Flashlight : MonoBehaviour
{
    public Transform flashlight;
    public float rotationOffset = -90f;
    void Update()
    {
        Vector3 mousePos = Camera.main.ScreenToWorldPoint(Input.mousePosition);
        Vector2 direction = (mousePos - transform.position).normalized;
        float angle = Mathf.Atan2(direction.y, direction.x) * Mathf.Rad2Deg;
        flashlight.rotation = Quaternion.Euler(0f, 0f, angle + rotationOffset);
    }
}
DeathAnimation.cs 
using System.Collections;
using TMPro;
using UnityEngine;

public class DeathAnimation : MonoBehaviour
{
    [SerializeField] PlayerHp playerHP;
    [SerializeField] GameObject player;
    [SerializeField] Vector3 lastPlayerLocation;
    [SerializeField] Vector3 playerLocation;
    [SerializeField] GameObject graveStone;
    int animationPlayedCount;
    [SerializeField] GameObject gameOver;
    [SerializeField] TextMeshProUGUI youDiedText;

    private void Update()
    {
        playerLocation = player.transform.position;
        if (playerHP.playerHp == 0 && animationPlayedCount != 1)
        {
            CreateGraveStone();
        }
    }
    public void CreateGraveStone()
    {
        animationPlayedCount++;
        lastPlayerLocation = playerLocation;
        Vector3 newLocation = new Vector3(lastPlayerLocation.x, lastPlayerLocation.y, lastPlayerLocation.z);
        Instantiate(graveStone, newLocation, Quaternion.Euler(0, 0, -20));
        player.SetActive(false);
        youDiedText.text = "You Died!";
        StartCoroutine(WaitTime());
    }
    IEnumerator WaitTime()
    {
        yield return new WaitForSeconds(1.5f);
        youDiedText.gameObject.SetActive(false);
        gameOver.SetActive(true);
    }
}
